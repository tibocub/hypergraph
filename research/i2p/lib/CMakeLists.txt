# Build: see build.cmd. NODE_LIB is a tiny import lib with only the napi_* symbols (node.lib would clash with our OpenSSL).
cmake_minimum_required(VERSION 3.20)
project(i2pd_addon CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
set(CMAKE_PREFIX_PATH ${DEPS})
find_package(Boost REQUIRED COMPONENTS atomic filesystem program_options)
find_package(OpenSSL REQUIRED)
find_package(ZLIB REQUIRED)
file(GLOB I2P_SRC ${I2PD_SRC}/libi2pd/*.cpp ${I2PD_SRC}/libi2pd_client/*.cpp ${I2PD_SRC}/i18n/*.cpp)
add_library(i2pd_addon SHARED addon.cpp ${I2P_SRC})
set_target_properties(i2pd_addon PROPERTIES PREFIX "" SUFFIX ".node")
target_include_directories(i2pd_addon PRIVATE ${I2PD_SRC}/libi2pd ${I2PD_SRC}/libi2pd_client ${I2PD_SRC}/i18n ${NODE_INC})
target_compile_definitions(i2pd_addon PRIVATE NO_TORRENTS WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0601 BOOST_ASIO_DISABLE_CONCEPTS)
target_compile_options(i2pd_addon PRIVATE /EHsc /utf-8 /bigobj /MP)
target_link_libraries(i2pd_addon PRIVATE ${NODE_LIB} Boost::atomic Boost::filesystem Boost::program_options OpenSSL::SSL OpenSSL::Crypto ZLIB::ZLIB ws2_32 iphlpapi crypt32 mswsock bcrypt)
